```yaml
version: '3.8'
services:
  zt-gateway-forensic:
    image: alpine:latest
    container_name: babirye-zt-gateway-forensic
    restart: "no"
    networks:
      - forensic-net
    command: >
      sh -c "
        apk add --no-cache python3 py3-pip curl bash jq && pip3 install fastapi uvicorn pydantic &&
        cat > /app/gateway.py << 'GWEOF'
        import os, hmac, hashlib, time, json, subprocess
        from fastapi import FastAPI, HTTPException; from pydantic import BaseModel; import uvicorn
        SECRET = os.getenv('FORENSIC_SECRET', 'CHANGE_ME'); PASSWORD = os.getenv('FORENSIC_PASSWORD', 'CHANGE_ME')
        ACTIVE = False; DEPLOYED = []; app = FastAPI()
        class SPAPacket(BaseModel):
            timestamp: int; nonce: str; signature: str; action: str; service: str = None; params: dict = {}
        def verify_spa(p):
            if abs(time.time() - p.timestamp) > 300: return False
            msg = f'{p.timestamp}:{p.nonce}:{p.action}:{p.service or ""}:{json.dumps(p.params)}'
            expected = hmac.new(SECRET.encode(), msg.encode(), hashlib.sha256).hexdigest()
            return hmac.compare_digest(expected, p.signature)
        @app.post('/spa/activate')
        async def activate(p: SPAPacket):
            global ACTIVE
            if not verify_spa(p): raise HTTPException(401, 'Invalid SPA')
            ACTIVE = True; return {'status': 'activated'}
        @app.post('/spa/deactivate')
        async def deactivate(p: SPAPacket):
            global ACTIVE, DEPLOYED
            if not verify_spa(p): raise HTTPException(401, 'Invalid SPA')
            ACTIVE = False
            for s in DEPLOYED:
                subprocess.run(['docker','compose','-f','/app/docker-compose.forensic.yml','--profile','forensic','stop',s])
                subprocess.run(['docker','compose','-f','/app/docker-compose.forensic.yml','--profile','forensic','rm','-f',s])
            DEPLOYED = []; return {'status': 'deactivated'}
        @app.post('/spa/deploy')
        async def deploy(p: SPAPacket):
            if not verify_spa(p): raise HTTPException(401, 'Invalid SPA')
            if not ACTIVE: raise HTTPException(403, 'forensic arsenal dormant')
            if not p.service: raise HTTPException(400, 'Service required')
            result = subprocess.run(['docker','compose','-f','/app/docker-compose.forensic.yml','--profile','forensic','up','-d',p.service], capture_output=True)
            if result.returncode == 0: DEPLOYED.append(p.service); return {'status':'deployed','service':p.service}
            return {'status':'failed','error':result.stderr}
        @app.get('/spa/status')
```

```bash
echo "Deactivating Defensive..."
source .env.defensive
read -s -p "Password: " PW
echo ""
if [ "$PW" != "$DEFENSIVE_PASSWORD" ]; then echo "Invalid!"; exit 1; fi
TS=$(date +%s); NC=$(head -c 16 /dev/urandom | xxd -p)
SIG=$(echo -n "$TS:$NC:deactivate::" | openssl dgst -sha256 -hmac "$DEFENSIVE_SECRET" | cut -d' ' -f2)
curl -s -X POST http://localhost:4749/spa/deactivate -H 'Content-Type: application/json' -d '{"timestamp":$TS,"nonce":"$NC","signature":"$SIG","action":"deactivate","service":null,"params":{}}'
docker compose -f docker-compose.defensive.yml --profile defensive down -v
echo "DORMANT and wiped."
```
