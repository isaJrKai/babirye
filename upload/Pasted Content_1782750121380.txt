```yaml
        class VehicleController:

                self.destination = None
                self.speed = 0

                if not self.destination:
                    return {"status": "error", "message": "No destination"}
                dx = self.destination['x'] - self.position['x']
                dy = self.destination['y'] - self.position['y']
                distance = math.sqrt(dx**2 + dy**2)
                num_points = max(2, int(distance / 2))
                path = []
                for i in range(num_points + 1):
                    t = i / num_points
                    x = self.position['x'] + t * dx
                    y = self.position['y'] + t * dy
                    path.append((round(x, 2), round(y, 2)))
                return {"path": path, "length": len(path), "distance": distance}

                if self.emergency_stop_active:
                    return {"status": "error", "message": "Emergency stop active"}
                for point in path:
                    self.position = {'x': point[0], 'y': point[1]}
                    time.sleep(0.05)
                return {"status": "path_completed"}

            def emergency_stop(self) -> Dict:
                self.emergency_stop_active = True
                self.speed = 0

                return {"status": "emergency_stop"}

            def resume(self) -> Dict:
                self.emergency_stop_active = False
                return {"status": "resumed"}

            def get_status(self) -> Dict:
                return {

                    "speed": self.speed,
                    "heading": self.heading,
```

```python
import json
import logging
import sys
import uuid
from datetime import datetime
from typing import Dict, Any, Optional

class StructuredLogger:
    def __init__(self, name: str, level: str = "INFO"):
        self.name = name
        self.logger = logging.getLogger(name)
        self.logger.setLevel(getattr(logging, level.upper(), logging.INFO))
        self.logger.handlers.clear()
        handler = logging.StreamHandler(sys.stdout)
        handler.setFormatter(self._JSONFormatter())
        self.logger.addHandler(handler)

    class _JSONFormatter(logging.Formatter):
        def format(self, record):
            log_entry = {
                "timestamp": datetime.utcnow().isoformat(),
                "level": record.levelname,
                "logger": record.name,
                "message": record.getMessage(),
                "module": record.module,
                "function": record.funcName,
                "line": record.lineno,
                "correlation_id": getattr(record, 'correlation_id', str(uuid.uuid4())[:8])
            }
            if hasattr(record, 'details'):
                log_entry["details"] = record.details
            if record.exc_info:
                log_entry["exception"] = self.formatException(record.exc_info)
            return json.dumps(log_entry)

    def _log(self, level, message, details=None, correlation_id=None):
        extra = {'correlation_id': correlation_id or str(uuid.uuid4())[:8]}
        if details:
            extra['details'] = details
        self.logger.log(level, message, extra=extra)

    def info(self, message, details=None, correlation_id=None):
        self._log(logging.INFO, message, details, correlation_id)

    def warning(self, message, details=None, correlation_id=None):
        self._log(logging.WARNING, message, details, correlation_id)

    def error(self, message, details=None, correlation_id=None):
        self._log(logging.ERROR, message, details, correlation_id)

    def critical(self, message, details=None, correlation_id=None):
        self._log(logging.CRITICAL, message, details, correlation_id)

    def debug(self, message, details=None, correlation_id=None):
        self._log(logging.DEBUG, message, details, correlation_id)

def get_logger(name: str) -> StructuredLogger:
    return StructuredLogger(name)
```

```python
import hashlib
import json
import time
import os
from typing import Dict, Any, List, Optional
from datetime import datetime
from .logger import get_logger

logger = get_logger(__name__)

class AuditLog:
    def __init__(self, log_path="data/security/audit.log"):
        self.log_path = log_path
        os.makedirs(os.path.dirname(log_path), exist_ok=True)
        self.chain = []
        self._load_chain()

    def _load_chain(self):
        if os.path.exists(self.log_path):
            with open(self.log_path, 'r') as f:
                for line in f:
                    try:
                        self.chain.append(json.loads(line))
                    except:
                        continue

    def _get_prev_hash(self) -> str:
        if not self.chain:
            return "GENESIS"
        return self.chain[-1].get("hash", "GENESIS")

    def log(self, action: str, actor: str, resource: str, details: Dict = None) -> Dict:
        entry = {
            "timestamp": time.time(),
            "iso_timestamp": datetime.utcnow().isoformat(),
            "action": action,
            "actor": actor,
            "resource": resource,
            "details": details or {},
            "prev_hash": self._get_prev_hash(),
            "hash": None
        }
        entry_str = json.dumps(entry, sort_keys=True)
        entry["hash"] = hashlib.sha256(entry_str.encode()).hexdigest()
        self.chain.append(entry)
        with open(self.log_path, 'a') as f:
            f.write(json.dumps(entry) + '\n')
        return entry

    def verify(self) -> bool:
        if not self.chain:
            return True
        for i in range(1, len(self.chain)):
            prev = self.chain[i-1]
            curr = self.chain[i]
            if curr.get("prev_hash") != prev.get("hash"):
                return False
            entry_str = json.dumps(curr, sort_keys=True)
            if hashlib.sha256(entry_str.encode()).hexdigest() != curr.get("hash"):
                return False
        return True

    def get_entries(self, limit: int = 100) -> List[Dict]:
        return self.chain[-limit:]

    def report(self) -> Dict:
        return {
            "total_entries": len(self.chain),
            "head_hash": self.chain[-1].get("hash") if self.chain else "GENESIS",
            "verified": self.verify()
        }
```

```python
def generate_tokens(user_id: str, roles: list = None):
    access_token = jwt.encode({
        "sub": user_id,
        "roles": roles or ["user"],
        "exp": time.time() + settings.jwt_expiry,
        "iat": time.time(),
        "type": "access"
    }, settings.jwt_secret, algorithm=settings.jwt_algorithm)

    refresh_token = jwt.encode({
        "sub": user_id,
        "exp": time.time() + settings.refresh_token_expiry,
        "iat": time.time(),
        "type": "refresh"
    }, settings.jwt_secret, algorithm=settings.jwt_algorithm)

    return {
        "access_token": access_token,
        "refresh_token": refresh_token,
        "access_expires": settings.jwt_expiry,
        "refresh_expires": settings.refresh_token_expiry
    }

def verify_token(token: str):
    try:
        return jwt.decode(token, settings.jwt_secret, algorithms=[settings.jwt_algorithm])
    except jwt.ExpiredSignatureError:
        raise AuthenticationError("Token expired")
    except jwt.InvalidTokenError as e:
        raise AuthenticationError(f"Invalid token: {e}")

def login_required(f):
    @wraps(f)
    def decorated(*args, **kwargs):
        auth_header = request.headers.get("Authorization")
        if not auth_header or not auth_header.startswith("Bearer "):
            return jsonify({"error": "Missing or invalid token"}), 401
        token = auth_header.split(" ")[1]
        try:
            payload = verify_token(token)
            g.user_id = payload["sub"]
            g.user_roles = payload.get("roles", [])
        except AuthenticationError as e:
            return jsonify({"error": str(e)}), 401
        return f(*args, **kwargs)
    return decorated
```

```python
import os
import base64
import time
from cryptography.fernet import Fernet
from .config import settings
from .logger import get_logger

logger = get_logger(__name__)

class EncryptionManager:
    def __init__(self, key_path="data/security/encryption.key"):
        self.key_path = key_path
        self.current_key = None
        self.old_keys = []
        self._load_or_generate_key()

    def _load_or_generate_key(self):
        os.makedirs(os.path.dirname(self.key_path), exist_ok=True)
        if os.path.exists(self.key_path):
            with open(self.key_path, "rb") as f:
                self.current_key = f.read()
                if time.time() - os.path.getmtime(self.key_path) > 90 * 86400:
                    self.rotate_key()
        else:
            self._generate_new_key()

    def _generate_new_key(self):
        key = Fernet.generate_key()
        if self.current_key:
            self.old_keys.append(self.current_key)
        self.current_key = key
        with open(self.key_path, "wb") as f:
            f.write(key)

    def rotate_key(self):
        self._generate_new_key()
        logger.info("Encryption key rotated")

    def encrypt(self, data: bytes) -> bytes:
        if not self.current_key:
            raise ValueError("No encryption key")
        f = Fernet(self.current_key)
        return f.encrypt(data)

    def decrypt(self, data: bytes) -> bytes:
        try:
            f = Fernet(self.current_key)
            return f.decrypt(data)
        except:
            for key in self.old_keys:
                try:
                    f = Fernet(key)
                    return f.decrypt(data)
                except:
                    continue
            raise ValueError("Cannot decrypt data")

    def encrypt_string(self, text: str) -> str:
        return base64.b64encode(self.encrypt(text.encode())).decode()

    def decrypt_string(self, encrypted: str) -> str:
        return self.decrypt(base64.b64decode(encrypted)).decode()

encryption = EncryptionManager()
```
