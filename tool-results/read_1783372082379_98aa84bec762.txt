    71→    71→

    77→    77→  // Token management

   175→   175→      const res = await fetch("/api/auth", {
   176→   176→        method: "POST",
   177→   177→        headers: { "Content-Type": "application/json", Authorization: `Bearer ${localStorage.getItem("babirye_token")}` },

   194→   194→      method: "DELETE",

   280→   280→            </Card>

   283→   283→
   284→   284→        {/* Users Tab */}

   317→   317→                                <SelectItem value="operator">Operator</SelectItem>

   341→   341→                        </TableCell>
   342→   342→                      </TableRow>
   343→   343→                    ))}
   344→   344→                  </TableBody>
   345→   345→                </Table>
   346→   346→              </ScrollArea>
   347→   347→            </CardContent>
   348→   348→          </Card>
   349→   349→        </TabsContent>
   350→   350→
   351→   351→        {/* API Tokens Tab */}
   352→   352→        <TabsContent value="tokens">
   353→   353→          <div className="flex justify-end mb-3">
   354→   354→            <Button size="sm" className="gap-1.5 text-xs" onClick={() => { setCreateTokenOpen(true); setCreatedToken(null); }}>
   355→   355→              <Plus className="h-3.5 w-3.5" />Create Token
