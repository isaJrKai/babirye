   383→    # ── Static discovery methods ──────────────────────────────────────
   384→
   385→    @staticmethod
   386→    def list_potentials(category: Optional[str] = None) -> Dict[str, Any]:
   387→        """List potentials, optionally filtered by category.
   388→
   389→        Args:
   390→            category: If provided, only return potentials in this category.
   391→                      One of the 5 category keys, or None for all.
   392→        """
   393→        if category:
   394→            if category not in PotentialsEngine.CATEGORY_COUNTS:
   395→                return {
   396→                    "status": "error",
   397→                    "message": f"Unknown category: {category}",
   398→                    "available_categories": list(PotentialsEngine.CATEGORY_COUNTS.keys()),
   399→                }
   400→            filtered = {
   401→                k: v for k, v in PotentialsEngine.POTENTIALS.items()
   402→                if v["category"] == category
   403→            }
   404→            return {
   405→                "status": "ok",
   406→                "category": category,
   407→                "potentials": filtered,
   408→                "count": len(filtered),
   409→            }
   410→        return {
   411→            "status": "ok",
   412→            "total_potentials": PotentialsEngine.TOTAL,
   413→            "categories": PotentialsEngine.CATEGORY_COUNTS,
   414→            "potentials": PotentialsEngine.POTENTIALS,
   415→        }
   416→
   417→    @staticmethod
   418→    def get_potential(name: str) -> Dict[str, Any]:
   419→        """Return details for a single potential by name."""
   420→        pot = PotentialsEngine.POTENTIALS.get(name)
   421→        if not pot:
   422→            return {"status": "error", "message": f"Unknown potential: {name}"}
   423→        return {"status": "ok", "name": name, **pot}
   424→
   425→    @staticmethod
   426→    def search(query: str, category: Optional[str] = None, limit: int = 20) -> Dict[str, Any]:
   427→        """Search potentials by keyword in name or description.
   428→
   429→        Args:
   430→            query: Search string (case-insensitive substring match).
   431→            category: Optional category filter.
   432→            limit: Maximum results to return.
   433→        """
   434→        query_lower = query.lower()
   435→        results = {}
   436→        for name, info in PotentialsEngine.POTENTIALS.items():
   437→            if category and info["category"] != category:
   438→                continue
   439→            if query_lower in name.lower() or query_lower in info["description"].lower():
   440→                results[name] = info
   441→                if len(results) >= limit:
   442→                    break
   443→        return {
   444→            "status": "ok",
   445→            "query": query,
   446→            "category": category,
   447→            "results": results,
   448→            "count": len(results),
   449→        }
   450→
   451→    @staticmethod
   452→    def get_categories() -> Dict[str, Any]:
   453→        """Return all categories with their potential counts."""
   454→        return {
   455→            "status": "ok",
   456→            "total_potentials": PotentialsEngine.TOTAL,
   457→            "categories": PotentialsEngine.CATEGORY_COUNTS,
   458→        }
   459→
   460→    # ── Instance exploration / activation ─────────────────────────────
   461→
   462→    def explore(self, query: str, max_results: int = 10) -> Dict[str, Any]:
   463→        """Explore potentials matching a query (instance method, logs exploration)."""
   464→        logger.info(f"Potentials exploration: query='{query}' max={max_results}")
   465→        search_result = self.search(query, limit=max_results)
   466→        entry = {
   467→            "query": query,
   468→            "max_results": max_results,
   469→            "results_count": search_result["count"],
   470→            "timestamp": __import__("datetime").datetime.now(
   471→                __import__("datetime").timezone.utc
   472→            ).isoformat(),
   473→        }
   474→        self._exploration_log.append(entry)
   475→        return {
   476→            "status": "ok",
   477→            "query": query,
   478→            "total_available": self.TOTAL,
   479→            "categories": self.CATEGORY_COUNTS,
   480→            **search_result,
   481→        }
   482→
   483→    def activate(self, potential_name: str, require_council: bool = True) -> Dict[str, Any]:
   484→        """Activate a potential capability. Council approval required by default.
   485→
   486→        Args:
   487→            potential_name: The slug of the potential to activate.
   488→            require_council: If True (default), the expert council must approve.
   489→        """
   490→        pot = PotentialsEngine.POTENTIALS.get(potential_name)
   491→        if not pot:
   492→            return {"status": "error", "message": f"Unknown potential: {potential_name}"}
   493→
   494→        if potential_name in self._activated:
   495→            return {"status": "ok", "message": f"Potential '{potential_name}' is already active", "total_activated": len(self._activated)}
   496→
   497→        if require_council:
   498→            decision = expert_council.consult(
   499→                f"Activate potential: {potential_name}",
   500→                f"Category: {pot['category']} | Description: {pot['description']} | Currently activated: {len(self._activated)}",
   501→            )
   502→            if decision.decision != "approved":
   503→                logger.warning(f"Council denied potential activation: {potential_name}")
   504→                return {"status": "denied", "potential": potential_name, "council": decision.to_dict()}
   505→        self._activated.append(potential_name)
   506→        logger.info(f"Activated potential: {potential_name}")
   507→        return {"status": "activated", "potential": potential_name, "category": pot["category"], "total_activated": len(self._activated)}
   508→
   509→    def deactivate(self, potential_name: str) -> Dict[str, Any]:
   510→        """Deactivate a previously activated potential."""
   511→        if potential_name in self._activated:
   512→            self._activated.remove(potential_name)
   513→            logger.info(f"Deactivated potential: {potential_name}")
   514→            return {"status": "deactivated", "potential": potential_name, "total_activated": len(self._activated)}
   515→        return {"status": "ok", "message": f"Potential '{potential_name}' was not active"}
   516→
   517→    def status(self) -> Dict[str, Any]:
   518→        """Return the current engine status including activated potentials."""
   519→        activated_details = {
   520→            name: PotentialsEngine.POTENTIALS[name]
   521→            for name in self._activated
   522→            if name in PotentialsEngine.POTENTIALS
   523→        }
   524→        return {
   525→            "total_potentials": PotentialsEngine.TOTAL,
   526→            "categories": PotentialsEngine.CATEGORY_COUNTS,
   527→            "activated_count": len(self._activated),
   528→            "activated": self._activated,
   529→            "activated_details": activated_details,
   530→        }
