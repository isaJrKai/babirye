   420→        def verify_spa(p):
   421→            if abs(time.time() - p.timestamp) > 60: return False
   422→            msg = f'{p.timestamp}:{p.nonce}:{p.action}:{p.service or ""}:{json.dumps(p.params)}'

   424→            return hmac.compare_digest(expected, p.signature)
   425→
   426→        @app.post('/spa/activate')
   427→        async def activate(p: SPAPacket):
   428→            global ACTIVE
   429→            if not verify_spa(p): raise HTTPException(401, 'Invalid SPA')
   430→            ACTIVE = True
   431→            return {'status': 'activated'}
   432→
   433→        @app.post('/spa/deactivate')

   441→            return {'status': 'deactivated'}
   442→
   443→        @app.post('/spa/deploy')

   449→            if result.returncode == 0:
   450→                DEPLOYED.append(p.service)
   451→                return {'status': 'deployed', 'service': p.service}
   452→            return {'status': 'failed', 'error': result.stderr}
   453→
   454→        @app.get('/spa/status')
   455→        async def status():
   456→            return {'active': ACTIVE, 'deployed': DEPLOYED}
   457→
   458→        @app.get('/spa/services')

   661→            action: str
   662→            service: str = None
   663→            params: dict = {}
   664→
   665→        def verify_spa(p):
   666→            if abs(time.time() - p.timestamp) > 60: return False
   667→            msg = f'{p.timestamp}:{p.nonce}:{p.action}:{p.service or ""}:{json.dumps(p.params)}'
   668→            expected = hmac.new(SECRET.encode(), msg.encode(), hashlib.sha256).hexdigest()
   669→            return hmac.compare_digest(expected, p.signature)
   670→
   671→        @app.post('/spa/activate')
   672→        async def activate(p: SPAPacket):

  1322→import hashlib, os, json, time, hmac, subprocess
  1323→from cryptography.fernet import Fernet
  1324→from ..core.logger import get_logger

  1328→class SecurityManager:
  1329→    def __init__(self):
  1330→        self.integrity_db = {}
  1331→        self.audit_chain = []
  1332→        self._load_key()
  1333→        self._scan_integrity()
  1334→
  1335→    def _load_key(self):
  1336→        key_file = "/app/security/encryption.key"
  1337→        if os.path.exists(key_file):
  1338→            with open(key_file, "rb") as f:
  1339→                self.key = f.read()
  1340→        else:
  1341→            self.key = Fernet.generate_key()
  1342→            os.makedirs("/app/security", exist_ok=True)
  1343→            with open(key_file, "wb") as f:
  1344→                f.write(self.key)
  1345→        self.f = Fernet(self.key)
  1346→
  1347→    def _scan_integrity(self):
  1348→        paths = ["/app/src/core/config.py", "/app/constitution/HAL_Constitution.md"]
  1349→        for p in paths:
  1350→            if os.path.exists(p):
  1351→                with open(p, "rb") as f:
  1352→                    self.integrity_db[p] = hashlib.sha256(f.read()).hexdigest()
  1353→
  1354→    def verify_integrity(self):
  1355→        results = {}
  1356→        for p, expected in self.integrity_db.items():
  1357→            if os.path.exists(p):
  1358→                with open(p, "rb") as f:
  1359→                    results[p] = hashlib.sha256(f.read()).hexdigest() == expected
  1360→            else:
  1361→                results[p] = False
  1362→        return results
  1363→
  1364→    def audit_log(self, action, details):
  1365→        entry = {
  1366→            "timestamp": time.time(),
  1367→            "action": action,
  1368→            "details": details,
  1369→            "prev_hash": self.audit_chain[-1].get("hash") if self.audit_chain else "GENESIS"
  1370→        }
  1371→        entry["hash"] = hashlib.sha256(json.dumps(entry, sort_keys=True).encode()).hexdigest()
  1372→        self.audit_chain.append(entry)
  1373→        with open("/app/security/audit.log", "a") as f:
  1374→            f.write(json.dumps(entry) + "\n")
  1375→
  1376→    def verify_password(self, password):
  1377→        expected = os.getenv("FORENSIC_PASSWORD", "CHANGE_ME")
  1378→        return hmac.compare_digest(password, expected)
  1379→
  1380→security_manager = SecurityManager()
