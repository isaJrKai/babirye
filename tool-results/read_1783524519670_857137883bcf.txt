     1→/**
     2→ * BABIRYE v10.0.0 — VIP Protection API Routes
     3→ * Profiles, drones, vehicles, wearables, incidents, and SOS triggering.
     4→ */
     5→import { NextRequest, NextResponse } from 'next/server';
     6→import { db } from '@/lib/db';
     7→import { authenticateRequest } from '@/lib/auth';
     8→
     9→// ---------------------------------------------------------------------------
    10→// GET — List VIP data by ?view= param
    11→//   profiles  : all VIP profiles with counts (drones/vehicles/wearables/incidents)
    12→//   drones    : all drones
    13→//   vehicles  : all vehicles
    14→//   wearables : all wearables
    15→//   incidents : all incidents
    16→// ---------------------------------------------------------------------------
    17→export async function GET(request: NextRequest) {
    18→  try {
    19→    const auth = await authenticateRequest(request.headers.get('Authorization'));
    20→    if (!auth.authenticated || !auth.payload) {
    21→      return NextResponse.json({ error: 'Authentication required' }, { status: 401 });
    22→    }
    23→
    24→    const { searchParams } = new URL(request.url);
    25→    const view = searchParams.get('view') || 'profiles';
    26→
    27→    switch (view) {
    28→      // ---- Profiles with counts ------------------------------------------------
    29→      case 'profiles': {
    30→        const profiles = await db.vipProfile.findMany({
    31→          orderBy: { createdAt: 'desc' },
    32→          include: {
    33→            _count: {
    34→              select: { drones: true, vehicles: true, wearables: true, incidents: true },
    35→            },
    36→          },
    37→        });
    38→        const result = profiles.map((p) => ({
    39→          id: p.id,
    40→          name: p.name,
    41→          designation: p.designation,
    42→          threatLevel: p.threatLevel,
    43→          protectionLevel: p.protectionLevel,
    44→          status: p.status,
    45→          bleTagId: p.bleTagId,
    46→          emergencyContact: p.emergencyContact,
    47→          createdAt: p.createdAt,
    48→          updatedAt: p.updatedAt,
    49→          droneCount: p._count.drones,
    50→          vehicleCount: p._count.vehicles,
    51→          wearableCount: p._count.wearables,
    52→          incidentCount: p._count.incidents,
    53→        }));
    54→        return NextResponse.json({ profiles: result, total: result.length });
    55→      }
    56→
    57→      // ---- Drones --------------------------------------------------------------
    58→      case 'drones': {
    59→        const vipId = searchParams.get('vipId');
    60→        const where = vipId ? { vipId } : {};
    61→        const drones = await db.vipDrone.findMany({
    62→          where,
    63→          orderBy: { createdAt: 'desc' },
    64→        });
    65→        return NextResponse.json({ drones, total: drones.length });
    66→      }
    67→
    68→      // ---- Vehicles ------------------------------------------------------------
    69→      case 'vehicles': {
    70→        const vipId = searchParams.get('vipId');
    71→        const where = vipId ? { vipId } : {};
    72→        const vehicles = await db.vipVehicle.findMany({
    73→          where,
    74→          orderBy: { createdAt: 'desc' },
    75→        });
    76→        return NextResponse.json({ vehicles, total: vehicles.length });
    77→      }
    78→
    79→      // ---- Wearables -----------------------------------------------------------
    80→      case 'wearables': {
    81→        const vipId = searchParams.get('vipId');
    82→        const where = vipId ? { vipId } : {};
    83→        const wearables = await db.vipWearable.findMany({
    84→          where,
    85→          orderBy: { createdAt: 'desc' },
    86→        });
    87→        return NextResponse.json({ wearables, total: wearables.length });
    88→      }
    89→
    90→      // ---- Incidents -----------------------------------------------------------
    91→      case 'incidents': {
    92→        const vipId = searchParams.get('vipId');
    93→        const where = vipId ? { vipId } : {};
    94→        const incidents = await db.vipIncident.findMany({
    95→          where,
    96→          orderBy: { createdAt: 'desc' },
    97→        });
    98→        return NextResponse.json({ incidents, total: incidents.length });
    99→      }
   100→
   101→      default:
   102→        return NextResponse.json(
   103→          { error: 'Invalid view. Use: profiles, drones, vehicles, wearables, incidents' },
   104→          { status: 400 }
   105→        );
   106→    }
   107→  } catch (error) {
   108→    console.error('VIP GET error:', error);
   109→    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
   110→  }
   111→}

   113→// ---------------------------------------------------------------------------
   114→// POST — Create VIP entity based on ?type= param
   115→//   ?type=profile|drone|vehicle|wearable|incident
   116→//   ?action=trigger_sos — triggers SOS on a wearable
   117→// ---------------------------------------------------------------------------
   118→export async function POST(request: NextRequest) {
   119→  try {
   120→    const auth = await authenticateRequest(request.headers.get('Authorization'));
   121→    if (!auth.authenticated || !auth.payload) {
   122→      return NextResponse.json({ error: 'Authentication required' }, { status: 401 });
   123→    }
   124→
   125→    const { searchParams } = new URL(request.url);
   126→    const type = searchParams.get('type');
   127→    const action = searchParams.get('action');
   128→
   129→    // ---- SOS trigger ----------------------------------------------------------
   130→    if (action === 'trigger_sos') {
   131→      const body = await request.json();
   132→      const { wearableId } = body;
   133→
   134→      if (!wearableId) {
   135→        return NextResponse.json({ error: 'wearableId is required' }, { status: 400 });
   136→      }
   137→
   138→      const wearable = await db.vipWearable.findUnique({
   139→        where: { id: wearableId },
   140→        include: { vip: true },
   141→      });
   142→
   143→      if (!wearable) {
   144→        return NextResponse.json({ error: 'Wearable not found' }, { status: 404 });
   145→      }
   146→
   147→      const updated = await db.vipWearable.update({
   148→        where: { id: wearableId },
   149→        data: {
   150→          sosTriggered: true,
   151→          status: 'alert',
   152→        },
   153→      });
   154→
   155→      // Create an SOS incident
   156→      const incident = await db.vipIncident.create({
   157→        data: {
   158→          vipId: wearable.vipId,
   159→          type: 'sos',
   160→          severity: 'critical',
   161→          description: `SOS triggered via ${wearable.deviceName} (${wearable.type})`,
   162→        },
   163→      });
   164→
   165→      // Create a notification
   166→      await db.notification.create({
   167→        data: {
   168→          type: 'vip',
   169→          title: `SOS Alert — ${wearable.vip.name}`,
   170→          message: `SOS triggered by ${wearable.deviceName} for ${wearable.vip.name}`,
   171→          severity: 'critical',
   172→          metadata: JSON.stringify({
   173→            vipId: wearable.vipId,
   174→            wearableId: wearable.id,
   175→            incidentId: incident.id,
   176→          }),
   177→        },
   178→      });
   179→
   180→      await db.auditLog.create({
   181→        data: {
   182→          userId: auth.payload.userId,
   183→          action: 'trigger_sos',
   184→          resource: 'vip_wearable',
   185→          details: JSON.stringify({ wearableId, vipId: wearable.vipId, incidentId: incident.id }),
   186→          statusCode: 200,
   187→        },
   188→      });
   189→
   190→      return NextResponse.json({ wearable: updated, incident });
   191→    }
   192→
   193→    // ---- Create entity by type ------------------------------------------------
   194→    if (!type) {
   195→      return NextResponse.json(
   196→        { error: '?type= param required (profile, drone, vehicle, wearable, incident)' },
   197→        { status: 400 }
   198→      );
   199→    }
   200→
   201→    const body = await request.json();
   202→
   203→    switch (type) {
   204→      // ---- Create VIP profile -------------------------------------------------
   205→      case 'profile': {
   206→        const { name, designation, threatLevel, protectionLevel, bleTagId, emergencyContact, medicalInfo, travelSchedule } = body;
   207→
   208→        if (!name || !designation) {
   209→          return NextResponse.json({ error: 'name and designation are required' }, { status: 400 });
   210→        }
   211→
   212→        const validThreat = ['low', 'medium', 'high', 'critical'];
   213→        const validProtection = ['standard', 'elevated', 'maximum'];
   214→
   215→        const profile = await db.vipProfile.create({
   216→          data: {
   217→            name,
   218→            designation,
   219→            threatLevel: validThreat.includes(threatLevel) ? threatLevel : 'medium',
   220→            protectionLevel: validProtection.includes(protectionLevel) ? protectionLevel : 'standard',
   221→            bleTagId: bleTagId || null,
   222→            emergencyContact: emergencyContact || null,
   223→            medicalInfo: medicalInfo || null,
   224→            travelSchedule: travelSchedule ? JSON.stringify(travelSchedule) : null,
   225→          },
   226→        });
   227→
   228→        await db.auditLog.create({
   229→          data: {
   230→            userId: auth.payload.userId,
   231→            action: 'create_vip_profile',
   232→            resource: 'vip_profile',
   233→            details: JSON.stringify({ profileId: profile.id, name }),
   234→            statusCode: 201,
   235→          },
   236→        });
   237→
   238→        return NextResponse.json({ profile }, { status: 201 });
   239→      }
   240→
   241→      // ---- Create drone -------------------------------------------------------
   242→      case 'drone': {
   243→        const { vipId, name, model, batteryLevel, cameraActive, thermalActive, mission } = body;
   244→
   245→        if (!vipId || !name || !model) {
   246→          return NextResponse.json({ error: 'vipId, name, and model are required' }, { status: 400 });
   247→        }
   248→
   249→        const vip = await db.vipProfile.findUnique({ where: { id: vipId } });
   250→        if (!vip) {
   251→          return NextResponse.json({ error: 'VIP profile not found' }, { status: 404 });
   252→        }
   253→
   254→        const drone = await db.vipDrone.create({
   255→          data: {
   256→            vipId,
   257→            name,
   258→            model,
   259→            batteryLevel: batteryLevel ?? 100,
   260→            cameraActive: cameraActive ?? false,
   261→            thermalActive: thermalActive ?? false,
   262→            mission: mission ? JSON.stringify(mission) : null,
   263→          },
   264→        });
   265→
   266→        await db.auditLog.create({
   267→          data: {
   268→            userId: auth.payload.userId,
   269→            action: 'create_vip_drone',
   270→            resource: 'vip_drone',
   271→            details: JSON.stringify({ droneId: drone.id, vipId, name }),
   272→            statusCode: 201,
   273→          },
   274→        });
   275→
   276→        return NextResponse.json({ drone }, { status: 201 });
   277→      }
   278→
   279→      // ---- Create vehicle -----------------------------------------------------
   280→      case 'vehicle': {
   281→        const { vipId, name, type, plate, driverId, route } = body;
   282→
   283→        if (!vipId || !name || !type) {
   284→          return NextResponse.json({ error: 'vipId, name, and type are required' }, { status: 400 });
   285→        }
   286→
   287→        const validVehicleTypes = ['sedan', 'suv', 'armored', 'motorcycle'];
   288→        if (!validVehicleTypes.includes(type)) {
   289→          return NextResponse.json(
   290→            { error: `Invalid type. Must be: ${validVehicleTypes.join(', ')}` },
   291→            { status: 400 }
   292→          );
   293→        }
   294→
   295→        const vip = await db.vipProfile.findUnique({ where: { id: vipId } });
   296→        if (!vip) {
   297→          return NextResponse.json({ error: 'VIP profile not found' }, { status: 404 });
   298→        }
   299→
   300→        const vehicle = await db.vipVehicle.create({
   301→          data: {
   302→            vipId,
   303→            name,
   304→            type,
   305→            plate: plate || null,
   306→            driverId: driverId || null,
   307→            route: route ? JSON.stringify(route) : null,
   308→          },
   309→        });
   310→
   311→        await db.auditLog.create({
   312→          data: {
   313→            userId: auth.payload.userId,
   314→            action: 'create_vip_vehicle',
   315→            resource: 'vip_vehicle',
   316→            details: JSON.stringify({ vehicleId: vehicle.id, vipId, name, type }),
   317→            statusCode: 201,
   318→          },
   319→        });
   320→
   321→        return NextResponse.json({ vehicle }, { status: 201 });
   322→      }
   323→
   324→      // ---- Create wearable ----------------------------------------------------
   325→      case 'wearable': {
   326→        const { vipId, type, deviceName, batteryLevel } = body;
   327→
   328→        if (!vipId || !type || !deviceName) {
   329→          return NextResponse.json({ error: 'vipId, type, and deviceName are required' }, { status: 400 });
   330→        }
   331→
   332→        const validWearableTypes = ['smartwatch', 'health_monitor', 'panic_button', 'gps_tracker'];
   333→        if (!validWearableTypes.includes(type)) {
   334→          return NextResponse.json(
   335→            { error: `Invalid type. Must be: ${validWearableTypes.join(', ')}` },
   336→            { status: 400 }
   337→          );
   338→        }
   339→
   340→        const vip = await db.vipProfile.findUnique({ where: { id: vipId } });
   341→        if (!vip) {
   342→          return NextResponse.json({ error: 'VIP profile not found' }, { status: 404 });
   343→        }
   344→
   345→        const wearable = await db.vipWearable.create({
   346→          data: {
   347→            vipId,
   348→            type,
   349→            deviceName,
   350→            batteryLevel: batteryLevel ?? 100,
   351→            lastPingAt: new Date(),
   352→          },
   353→        });
   354→
   355→        await db.auditLog.create({
   356→          data: {
   357→            userId: auth.payload.userId,
   358→            action: 'create_vip_wearable',
   359→            resource: 'vip_wearable',
   360→            details: JSON.stringify({ wearableId: wearable.id, vipId, type, deviceName }),
   361→            statusCode: 201,
   362→          },
   363→        });
   364→
   365→        return NextResponse.json({ wearable }, { status: 201 });
   366→      }
   367→
   368→      // ---- Create incident ----------------------------------------------------
   369→      case 'incident': {
   370→        const { vipId, type, severity, description, response } = body;
   371→
   372→        if (!vipId || !type || !description) {
   373→          return NextResponse.json({ error: 'vipId, type, and description are required' }, { status: 400 });
   374→        }
   375→
   376→        const validIncidentTypes = ['threat', 'medical', 'route_deviation', 'perimeter_breach', 'sos'];
   377→        const validSeverities = ['low', 'medium', 'high', 'critical'];
   378→
   379→        if (!validIncidentTypes.includes(type)) {
   380→          return NextResponse.json(
   381→            { error: `Invalid type. Must be: ${validIncidentTypes.join(', ')}` },
   382→            { status: 400 }
   383→          );
   384→        }
   385→
   386→        const vip = await db.vipProfile.findUnique({ where: { id: vipId } });
   387→        if (!vip) {
   388→          return NextResponse.json({ error: 'VIP profile not found' }, { status: 404 });
   389→        }
   390→
   391→        const incident = await db.vipIncident.create({
   392→          data: {
   393→            vipId,
   394→            type,
   395→            severity: validSeverities.includes(severity) ? severity : 'medium',
   396→            description,
   397→            response: response || null,
   398→          },
   399→        });
   400→
   401→        await db.auditLog.create({
   402→          data: {
   403→            userId: auth.payload.userId,
   404→            action: 'create_vip_incident',
   405→            resource: 'vip_incident',
   406→            details: JSON.stringify({ incidentId: incident.id, vipId, type, severity }),
   407→            statusCode: 201,
   408→          },
   409→        });
   410→
   411→        return NextResponse.json({ incident }, { status: 201 });
   412→      }
   413→
   414→      default:
   415→        return NextResponse.json(
   416→          { error: 'Invalid type. Use: profile, drone, vehicle, wearable, incident' },
   417→          { status: 400 }
   418→        );
   419→    }
   420→  } catch (error) {
   421→    console.error('VIP POST error:', error);
   422→    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
   423→  }
   424→}

   429→export async function PUT(request: NextRequest) {
   430→  try {
   431→    const auth = await authenticateRequest(request.headers.get('Authorization'));

   444→    }
   445→
   446→    const body = await request.json();
   447→    const { id, ...fields } = body;

   467→          return NextResponse.json({ error: 'No fields to update' }, { status: 400 });

   480→        });

   499→          return NextResponse.json({ error: 'No fields to update' }, { status: 400 });

   502→        const updated = await db.vipDrone.update({ where: { id }, data });

   529→

   531→          return NextResponse.json({ error: 'No fields to update' }, { status: 400 });
   532→        }
   533→
   534→        const updated = await db.vipVehicle.update({ where: { id }, data });
   535→
   536→        await db.auditLog.create({
   537→          data: {
   538→            userId: auth.payload.userId,

   541→            details: JSON.stringify({ id, fields: Object.keys(data) }),

   566→
   567→        await db.auditLog.create({
   568→          data: {
   569→            userId: auth.payload.userId,
   570→            action: 'update_vip_wearable',
   571→            resource: 'vip_wearable',
   572→            details: JSON.stringify({ id, fields: Object.keys(data) }),
   573→            statusCode: 200,
   574→          },
   575→        });
   576→
   577→        return NextResponse.json({ wearable: updated });
   578→      }
   579→
   580→      case 'incident': {
   581→        const data: Record<string, unknown> = {};
   582→        if (fields.type !== undefined) data.type = fields.type;
   583→        if (fields.severity !== undefined) data.severity = fields.severity;
   584→        if (fields.description !== undefined) data.description = fields.description;
   585→        if (fields.response !== undefined) data.response = fields.response;
   586→        if (fields.resolvedAt !== undefined) data.resolvedAt = fields.resolvedAt ? new Date(fields.resolvedAt) : null;
   587→
   588→        if (Object.keys(data).length === 0) {
   589→          return NextResponse.json({ error: 'No fields to update' }, { status: 400 });
   590→        }
   591→
   592→        const updated = await db.vipIncident.update({ where: { id }, data });
   593→
   594→        await db.auditLog.create({
   595→          data: {
   596→            userId: auth.payload.userId,
   597→            action: 'update_vip_incident',
   598→            resource: 'vip_incident',
   599→            details: JSON.stringify({ id, fields: Object.keys(data) }),
   600→            statusCode: 200,
   601→          },
   602→        });
   603→
   604→        return NextResponse.json({ incident: updated });
   605→      }
   606→
   607→      default:
   608→        return NextResponse.json(
   609→          { error: 'Invalid type. Use: profile, drone, vehicle, wearable, incident' },
   610→          { status: 400 }
   611→        );
   612→    }
   613→  } catch (error) {
   614→    console.error('VIP PUT error:', error);
   615→    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
   616→  }
   617→}

   619→// ---------------------------------------------------------------------------
   620→// DELETE — Remove VIP entity based on ?type= param
   621→// ---------------------------------------------------------------------------
   622→export async function DELETE(request: NextRequest) {
   623→  try {
   624→    const auth = await authenticateRequest(request.headers.get('Authorization'));
   625→    if (!auth.authenticated || !auth.payload) {
   626→      return NextResponse.json({ error: 'Authentication required' }, { status: 401 });
   627→    }
   628→
   629→    const { searchParams } = new URL(request.url);
   630→    const type = searchParams.get('type');
   631→    const id = searchParams.get('id');
   632→
   633→    if (!type || !id) {
   634→      return NextResponse.json({ error: '?type= and ?id= params are required' }, { status: 400 });
   635→    }
   636→
   637→    const validTypes = ['profile', 'drone', 'vehicle', 'wearable', 'incident'];
   638→    if (!validTypes.includes(type)) {
   639→      return NextResponse.json(
   640→        { error: `Invalid type. Use: ${validTypes.join(', ')}` },
   641→        { status: 400 }
   642→      );
   643→    }
   644→
   645→    switch (type) {
   646→      case 'profile': {
   647→        const existing = await db.vipProfile.findUnique({ where: { id } });
   648→        if (!existing) return NextResponse.json({ error: 'Profile not found' }, { status: 404 });
   649→        await db.vipProfile.delete({ where: { id } });
   650→        await db.auditLog.create({
   651→          data: { userId: auth.payload.userId, action: 'delete_vip_profile', resource: 'vip_profile', details: JSON.stringify({ id, name: existing.name }), statusCode: 200 },
   652→        });
   653→        return NextResponse.json({ success: true, message: 'VIP profile deleted' });
   654→      }
   655→
   656→      case 'drone': {
   657→        const existing = await db.vipDrone.findUnique({ where: { id } });
   658→        if (!existing) return NextResponse.json({ error: 'Drone not found' }, { status: 404 });

   660→        await db.auditLog.create({
   661→          data: { userId: auth.payload.userId, action: 'delete_vip_drone', resource: 'vip_drone', details: JSON.stringify({ id, name: existing.name }), statusCode: 200 },
   662→        });
   663→        return NextResponse.json({ success: true, message: 'Drone deleted' });
   664→      }
   665→
   666→      case 'vehicle': {
   667→        const existing = await db.vipVehicle.findUnique({ where: { id } });
   668→        if (!existing) return NextResponse.json({ error: 'Vehicle not found' }, { status: 404 });
   669→        await db.vipVehicle.delete({ where: { id } });
   670→        await db.auditLog.create({
   671→          data: { userId: auth.payload.userId, action: 'delete_vip_vehicle', resource: 'vip_vehicle', details: JSON.stringify({ id, name: existing.name }), statusCode: 200 },
   672→        });
   673→        return NextResponse.json({ success: true, message: 'Vehicle deleted' });
   674→      }
   675→
   676→      case 'wearable': {
   677→        const existing = await db.vipWearable.findUnique({ where: { id } });
   678→        if (!existing) return NextResponse.json({ error: 'Wearable not found' }, { status: 404 });
   679→        await db.vipWearable.delete({ where: { id } });

   681→          data: { userId: auth.payload.userId, action: 'delete_vip_wearable', resource: 'vip_wearable', details: JSON.stringify({ id, name: existing.deviceName }), statusCode: 200 },
   682→        });
   683→        return NextResponse.json({ success: true, message: 'Wearable deleted' });
   684→      }
   685→
   686→      case 'incident': {

   692→        });
   693→        return NextResponse.json({ success: true, message: 'Incident deleted' });
   694→      }

   697→    return NextResponse.json({ error: 'Unhandled type' }, { status: 400 });

   700→    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
