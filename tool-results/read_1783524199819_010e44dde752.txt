     9→     9→// ---------------------------------------------------------------------------
    10→    10→// GET — List VIP data by ?view= param
    11→    11→//   profiles  : all VIP profiles with counts (drones/vehicles/wearables/incidents)
    12→    12→//   drones    : all drones
    13→    13→//   vehicles  : all vehicles
    14→    14→//   wearables : all wearables
    15→    15→//   incidents : all incidents
    16→    16→// ---------------------------------------------------------------------------
    17→    17→export async function GET(request: NextRequest) {
    18→    18→  try {
    19→    19→    const auth = await authenticateRequest(request.headers.get('Authorization'));
    20→    20→    if (!auth.authenticated || !auth.payload) {
    21→    21→      return NextResponse.json({ error: 'Authentication required' }, { status: 401 });
    22→    22→    }
    23→    23→
    24→    24→    const { searchParams } = new URL(request.url);
    25→    25→    const view = searchParams.get('view') || 'profiles';
    26→    26→
    27→    27→    switch (view) {
    28→    28→      // ---- Profiles with counts ------------------------------------------------
    29→    29→      case 'profiles': {
    30→    30→        const profiles = await db.vipProfile.findMany({
    31→    31→          orderBy: { createdAt: 'desc' },
    32→    32→          include: {
    33→    33→            _count: {
    34→    34→              select: { drones: true, vehicles: true, wearables: true, incidents: true },
    35→    35→            },
    36→    36→          },
    37→    37→        });
    38→    38→        const result = profiles.map((p) => ({
    39→    39→          id: p.id,
    40→    40→          name: p.name,
    41→    41→          designation: p.designation,
    42→    42→          threatLevel: p.threatLevel,
    43→    43→          protectionLevel: p.protectionLevel,
    44→    44→          status: p.status,
    45→    45→          bleTagId: p.bleTagId,
    46→    46→          emergencyContact: p.emergencyContact,
    47→    47→          createdAt: p.createdAt,
    48→    48→          updatedAt: p.updatedAt,
    49→    49→          droneCount: p._count.drones,
    50→    50→          vehicleCount: p._count.vehicles,
    51→    51→          wearableCount: p._count.wearables,
    52→    52→          incidentCount: p._count.incidents,
    53→    53→        }));
    54→    54→        return NextResponse.json({ profiles: result, total: result.length });
    55→    55→      }
    56→    56→
    57→    57→      // ---- Drones --------------------------------------------------------------
    58→    58→      case 'drones': {
    59→    59→        const vipId = searchParams.get('vipId');
    60→    60→        const where = vipId ? { vipId } : {};
    61→    61→        const drones = await db.vipDrone.findMany({
    62→    62→          where,
    63→    63→          orderBy: { createdAt: 'desc' },
    64→    64→        });
    65→    65→        return NextResponse.json({ drones, total: drones.length });
    66→    66→      }
    67→    67→
    68→    68→      // ---- Vehicles ------------------------------------------------------------
    69→    69→      case 'vehicles': {
    70→    70→        const vipId = searchParams.get('vipId');
    71→    71→        const where = vipId ? { vipId } : {};
    72→    72→        const vehicles = await db.vipVehicle.findMany({
    73→    73→          where,
    74→    74→          orderBy: { createdAt: 'desc' },
    75→    75→        });
    76→    76→        return NextResponse.json({ vehicles, total: vehicles.length });
    77→    77→      }
    78→    78→
    79→    79→      // ---- Wearables -----------------------------------------------------------
    80→    80→      case 'wearables': {
    81→    81→        const vipId = searchParams.get('vipId');
    82→    82→        const where = vipId ? { vipId } : {};
    83→    83→        const wearables = await db.vipWearable.findMany({
    84→    84→          where,
    85→    85→          orderBy: { createdAt: 'desc' },
    86→    86→        });
    87→    87→        return NextResponse.json({ wearables, total: wearables.length });
    88→    88→      }
    89→    89→
    90→    90→      // ---- Incidents -----------------------------------------------------------
    91→    91→      case 'incidents': {
    92→    92→        const vipId = searchParams.get('vipId');
    93→    93→        const where = vipId ? { vipId } : {};
    94→    94→        const incidents = await db.vipIncident.findMany({
    95→    95→          where,
    96→    96→          orderBy: { createdAt: 'desc' },
    97→    97→        });
    98→    98→        return NextResponse.json({ incidents, total: incidents.length });
    99→    99→      }
   100→   100→
   101→   101→      default:
   102→   102→        return NextResponse.json(
   103→   103→          { error: 'Invalid view. Use: profiles, drones, vehicles, wearables, incidents' },
   104→   104→          { status: 400 }
   105→   105→        );
   106→   106→    }
   107→   107→  } catch (error) {
   108→   108→    console.error('VIP GET error:', error);
   109→   109→    return NextResponse.json({ error: 'Internal server error' }, { status: 500 });
   110→   110→  }
   111→   111→}

   424→   424→}

   427→   427→// PUT — Update VIP entity based on ?type= param
   428→   428→// ---------------------------------------------------------------------------
   429→   429→export async function PUT(request: NextRequest) {
   430→   430→  try {
   431→   431→    const auth = await authenticateRequest(request.headers.get('Authorization'));
   432→   432→    if (!auth.authenticated || !auth.payload) {
   433→   433→      return NextResponse.json({ error: 'Authentication required' }, { status: 401 });
   434→   434→    }
   435→   435→
   436→   436→    const { searchParams } = new URL(request.url);
   437→   437→    const type = searchParams.get('type');
   438→   438→
   439→   439→    if (!type) {
   440→   440→      return NextResponse.json(
   441→   441→        { error: '?type= param required (profile, drone, vehicle, wearable, incident)' },
   442→   442→        { status: 400 }
   443→   443→      );
   444→   444→    }
   445→   445→
   446→   446→    const body = await request.json();
   447→   447→    const { id, ...fields } = body;
   448→   448→
   449→   449→    if (!id) {
   450→   450→      return NextResponse.json({ error: 'Entity id is required' }, { status: 400 });
   451→   451→    }
   452→   452→
   453→   453→    switch (type) {
   454→   454→      case 'profile': {
   455→   455→        const data: Record<string, unknown> = {};
   456→   456→        if (fields.name !== undefined) data.name = fields.name;
   457→   457→        if (fields.designation !== undefined) data.designation = fields.designation;
   458→   458→        if (fields.threatLevel !== undefined) data.threatLevel = fields.threatLevel;
   459→   459→        if (fields.protectionLevel !== undefined) data.protectionLevel = fields.protectionLevel;
   460→   460→        if (fields.status !== undefined) data.status = fields.status;
   461→   461→        if (fields.bleTagId !== undefined) data.bleTagId = fields.bleTagId;
   462→   462→        if (fields.emergencyContact !== undefined) data.emergencyContact = fields.emergencyContact;
   463→   463→        if (fields.medicalInfo !== undefined) data.medicalInfo = fields.medicalInfo;
   464→   464→        if (fields.travelSchedule !== undefined) data.travelSchedule = JSON.stringify(fields.travelSchedule);
   465→   465→
   466→   466→        if (Object.keys(data).length === 0) {
   467→   467→          return NextResponse.json({ error: 'No fields to update' }, { status: 400 });
   468→   468→        }
   469→   469→
   470→   470→        const updated = await db.vipProfile.update({ where: { id }, data });
   471→   471→
   472→   472→        await db.auditLog.create({
   473→   473→          data: {
   474→   474→            userId: auth.payload.userId,
   475→   475→            action: 'update_vip_profile',
   476→   476→            resource: 'vip_profile',
   477→   477→            details: JSON.stringify({ id, fields: Object.keys(data) }),
   478→   478→            statusCode: 200,
   479→   479→          },
   480→   480→        });
   481→   481→
   482→   482→        return NextResponse.json({ profile: updated });
   483→   483→      }
   484→   484→
   485→   485→      case 'drone': {
   486→   486→        const data: Record<string, unknown> = {};
   487→   487→        if (fields.name !== undefined) data.name = fields.name;
   488→   488→        if (fields.model !== undefined) data.model = fields.model;
   489→   489→        if (fields.status !== undefined) data.status = fields.status;
   490→   490→        if (fields.batteryLevel !== undefined) data.batteryLevel = fields.batteryLevel;
   491→   491→        if (fields.altitude !== undefined) data.altitude = fields.altitude;
