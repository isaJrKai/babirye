   149→
   150→export async function seedDatabase(): Promise<{ success: boolean; message: string; stats: Record<string, number> }> {
   151→  const stats: Record<string, number> = {};
   152→
   153→  try {
   154→    // 1. Create admin user if none exists
   155→    const userCount = await db.user.count();
   156→    if (userCount === 0) {
   157→      const passwordHash = await hashPassword('Admin@BABIRYE2024!');
   158→      await db.user.create({
   159→        data: {
   160→          email: 'admin@babirye.local',
   161→          passwordHash,
   162→          name: 'System Administrator',
   163→          role: 'admin',
   164→          mfaEnabled: false,
   165→          isActive: true,
   166→        },
   167→      });
   168→      stats.users = 1;
   169→    } else {
   170→      stats.users = userCount;
   171→    }
   172→
   173→    // 2. Seed agents
   174→    const existingAgents = await db.agent.count();
   175→    if (existingAgents === 0) {
   176→      for (const agent of AGENTS) {
   177→        await db.agent.create({
   178→          data: {
   179→            ...agent,
   180→            config: JSON.stringify({ maxConcurrency: 5, timeout: 300, retryAttempts: 3 }),
   181→            metrics: JSON.stringify({ tasksCompleted: 0, tasksFailed: 0, avgResponseTime: 0, uptime: 99.9 }),
   182→            lastActivityAt: agent.status === 'running' ? new Date() : null,
   183→          },
   184→        });
   185→      }
   186→      stats.agents = AGENTS.length;
   187→    } else {
   188→      stats.agents = existingAgents;
   189→    }
   190→
   191→    // 3. Seed forensic tools
   192→    const existingForensic = await db.forensicTool.count();
   193→    if (existingForensic === 0) {
   194→      for (const tool of FORENSIC_TOOLS) {
   195→        await db.forensicTool.create({ data: tool });
   196→      }
   197→      stats.forensicTools = FORENSIC_TOOLS.length;
   198→    } else {
   199→      stats.forensicTools = existingForensic;
   200→    }
   201→
   202→    // 4. Seed offensive tools
   203→    const existingOffensive = await db.offensiveTool.count();
   204→    if (existingOffensive === 0) {
   205→      for (const tool of OFFENSIVE_TOOLS) {
   206→        await db.offensiveTool.create({ data: tool });
   207→      }
   208→      stats.offensiveTools = OFFENSIVE_TOOLS.length;
   209→    } else {
   210→      stats.offensiveTools = existingOffensive;
   211→    }
   212→
   213→    // 5. Seed defensive tools
   214→    const existingDefensive = await db.defensiveTool.count();
   215→    if (existingDefensive === 0) {
   216→      for (const tool of DEFENSIVE_TOOLS) {
   217→        await db.defensiveTool.create({ data: tool });
   218→      }
   219→      stats.defensiveTools = DEFENSIVE_TOOLS.length;
   220→    } else {
   221→      stats.defensiveTools = existingDefensive;
   222→    }
   223→
   224→    // 6. Seed micro-defensive tools
   225→    const existingMicroDef = await db.microDefensiveTool.count();
   226→    if (existingMicroDef === 0) {
   227→      for (const tool of MICRO_DEFENSIVE_TOOLS) {
   228→        await db.microDefensiveTool.create({
   229→          data: {
   230→            name: tool.name,
   231→            category: tool.category,
   232→            description: tool.description,
   233→            blockCount: tool.blockCount,
   234→          },
   235→        });
