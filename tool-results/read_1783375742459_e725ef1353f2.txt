    28→
    29→interface Setting {
    30→  id: string; key: string; value: string; category: string; isSecret: boolean; description: string | null;
    31→}
    32→interface UserRow {
    33→  id: string; email: string; name: string | null; role: string; isActive: boolean;
    34→  lastLoginAt: string | null; failedAttempts: number; lockedUntil: string | null; createdAt: string;
    35→}
    36→interface TokenRow {
    37→  id: string; name: string; purpose: string | null; isRevoked: boolean;
    38→  createdAt: string; expiresAt: string; lastUsedAt: string | null;
    39→}

    42→  general: "bg-slate-500/10 text-slate-400 border-slate-500/20",
    43→  security: "bg-red-500/10 text-red-400 border-red-500/20",
    44→  network: "bg-emerald-500/10 text-emerald-400 border-emerald-500/20",
    45→  api: "bg-amber-500/10 text-amber-400 border-amber-500/20",
    46→  notification: "bg-purple-500/10 text-purple-400 border-purple-500/20",
    47→  database: "bg-cyan-500/10 text-cyan-400 border-cyan-500/20",
    48→};
    49→
    50→const roleColors: Record<string, string> = {
    51→  admin: "bg-red-500/10 text-red-400 border-red-500/20",
    52→  operator: "bg-amber-500/10 text-amber-400 border-amber-500/20",
    53→  analyst: "bg-emerald-500/10 text-emerald-400 border-emerald-500/20",
    54→  viewer: "bg-slate-500/10 text-slate-400 border-slate-500/20",
    55→};
    56→
    57→const container = { hidden: { opacity: 0 }, show: { opacity: 1, transition: { staggerChildren: 0.04 } } };
    58→const item = { hidden: { opacity: 0, y: 8 }, show: { opacity: 1, y: 0, transition: { duration: 0.3 } } };
    59→
    60→export function SettingsPage() {
    61→  const { user } = useAppStore();
    62→  const { toast } = useToast();
    63→  const isAdmin = user?.role === "admin";
    64→
    65→  const { data: settings, loading: settingsLoading, refetch: refetchSettings } = useApi<Setting[]>("/api/settings");
    66→  const { data: users, loading: usersLoading, refetch: refetchUsers } = useApi<UserRow[]>("/api/auth?view=users");
    67→  const { data: tokens, loading: tokensLoading, refetch: refetchTokens } = useApi<TokenRow[]>("/api/auth?view=tokens");
    68→
    69→  const [revealedKeys, setRevealedKeys] = useState<Set<string>>(new Set());
    70→  const [editDialog, setEditDialog] = useState<{ open: boolean; setting: Setting | null }>({ open: false, editValue: "" });
    71→  const [editValue, setEditValue] = useState("");
    72→  const [saving, setSaving] = useState(false);
    73→
    74→  // User management
    75→  const [createUserOpen, setCreateUserOpen] = useState(false);
    76→  const [newUser, setNewUser] = useState({ name: "", email: "", password: "", role: "analyst" });
    77→  const [creatingUser, setCreatingUser] = useState(false);
    78→  const [deactivateTarget, setDeactivateTarget] = useState<UserRow | null>(null);
    79→  const [deactivatingUser, setDeactivatingUser] = useState(false);
    80→
    81→  // Token management
    82→  const [createTokenOpen, setCreateTokenOpen] = useState(false);
    83→  const [newToken, setNewToken] = useState({ name: "", purpose: "", ttl: "28800" });
    84→  const [createdToken, setCreatedToken] = useState<string | null>(null);
    85→  const [creatingToken, setCreatingToken] = useState(false);
    86→
    87→  // Security
    88→  const [passwordForm, setPasswordForm] = useState({ current: "", newPass: "", confirm: "" });
    89→  const [changingPassword, setChangingPassword] = useState(false);
    90→
    91→  const groupedSettings = useMemo(() => {
    92→    if (!settings) return {};
    93→    const groups: Record<string, Setting[]> = {};
    94→    for (const s of settings) {
    95→      if (!groups[s.category]) groups[s.category] = [];
    96→      groups[s.category].push(s);
    97→    }
    98→    return groups;
    99→  }, [settings]);
   100→
   101→  const toggleReveal = (key: string) => {
   102→    setRevealedKeys((prev) => {
   103→      const next = new Set(prev);
   104→      if (next.has(key)) next.delete(key);
   105→      else next.add(key);
   106→      return next;
   107→    });
   108→  };
   109→
   110→  const openEdit = (setting: Setting) => {
   111→    setEditDialog({ open: true, setting });
   112→    setEditValue(setting.value);
   113→  };
   114→
   115→  const saveSetting = async () => {
   116→    if (!editDialog.setting) return;
   117→    setSaving(true);
   118→    const res = await fetch("/api/settings", {
   119→      method: "PUT",
   120→      headers: { "Content-Type": "application/json", Authorization: `Bearer ${localStorage.getItem("babirye_token")}` },
   121→      body: JSON.stringify({ key: editDialog.setting.key, value: editValue }),
   122→    });
   123→    if (res.ok) {
   124→      toast({ title: "Setting updated", description: `${editDialog.setting.key} has been updated.` });
   125→      setEditDialog({ open: false, setting: null });
   126→      refetchSettings();
   127→    } else {
   128→      const d = await res.json().catch(() => ({}));
   129→      toast({ title: "Error", description: d.error || "Failed to update setting", variant: "destructive" });
   130→    }
   131→    setSaving(false);
   132→  };
   133→
   134→  const handleCreateUser = async () => {
   135→    if (!newUser.email || !newUser.password || !newUser.name) return;
   136→    setCreatingUser(true);
   137→    try {
   138→      const res = await fetch("/api/auth", {
   139→        method: "POST",
   140→        headers: { "Content-Type": "application/json", Authorization: `Bearer ${localStorage.getItem("babirye_token")}` },
   141→        body: JSON.stringify({ action: "create_user", ...newUser }),
   142→      });
   143→      if (res.ok) {
   144→        toast({ title: "User created", description: `${newUser.email} has been added.` });
   145→        setCreateUserOpen(false);
   146→        setNewUser({ name: "", email: "", password: "", role: "analyst" });
   147→        refetchUsers();
   148→      } else {
   149→        const d = await res.json().catch(() => ({}));
   150→        toast({ title: "Error", description: d.error || "Failed to create user", variant: "destructive" });
   151→      }
   152→    } catch { toast({ title: "Error", description: "Network error", variant: "destructive" }); }
   153→    setCreatingUser(false);
   154→  };
   155→
   156→  const handleRoleChange = async (userId: string, role: string) => {
   157→    await fetch("/api/auth", {
   158→      method: "PUT",
   159→      headers: { "Content-Type": "application/json", Authorization: `Bearer ${localStorage.getItem("babirye_token")}` },
   160→      body: JSON.stringify({ userId, role }),
   161→    });
   162→    refetchUsers();
   163→  };
   164→
   165→  const handleToggleActive = async (userId: string) => {
   166→    await fetch("/api/auth", {
   167→      method: "PUT",
   168→      headers: { "Content-Type": "application/json", Authorization: `Bearer ${localStorage.getItem("babirye_token")}` },
   169→      body: JSON.stringify({ userId, action: "toggle_active" }),
   170→    });
   171→    refetchUsers();
   172→  };
   173→
   174→  const handleCreateToken = async () => {
   175→    if (!newToken.name) return;
   176→    setCreatingToken(true);
   177→    setCreatedToken(null);
   178→    try {
   179→      const res = await fetch("/api/auth", {
   180→        method: "POST",
   181→        headers: { "Content-Type": "application/json", Authorization: `Bearer ${localStorage.getItem("babirye_token")}` },
   182→        body: JSON.stringify({ action: "create_token", name: newToken.name, purpose: newToken.purpose, ttl: parseInt(newToken.ttl, 10) }),
   183→      });
   184→      const data = await res.json();
   185→      if (res.ok) {
   186→        setCreatedToken(data.token);
   187→        setNewToken({ name: "", purpose: "", ttl: "28800" });
   188→        refetchTokens();
   189→      } else {
   190→        toast({ title: "Error", description: data.error || "Failed to create token", variant: "destructive" });
   191→      }
   192→    } catch { toast({ title: "Error", description: "Network error", variant: "destructive" }); }
   193→    setCreatingToken(false);
   194→  };
   195→
   196→  const handleRevokeToken = async (tokenId: string) => {
   197→    await fetch(`/api/auth?token_id=${tokenId}`, {
   198→      method: "DELETE",
   199→      headers: { Authorization: `Bearer ${localStorage.getItem("babirye_token")}` },
   200→    });
   201→    refetchTokens();
   202→    toast({ title: "Token revoked" });
   203→  };
   204→
   205→  const copyToClipboard = (text: string) => {
   206→    navigator.clipboard.writeText(text);
   207→    toast({ title: "Copied to clipboard" });
   208→  };
   209→
   210→  const fmtDate = (d: string | null) => {
   211→    if (!d) return "—";
   212→    return new Date(d).toLocaleString();
   213→  };
   214→
   215→  return (
   216→    <motion.div variants={container} initial="hidden" animate="show" className="space-y-6">
   217→      <motion.div variants={item} className="flex items-center justify-between">
   218→        <div>
   219→          <h1 className="text-2xl font-bold tracking-tight text-foreground">System Settings</h1>
   220→          <p className="text-sm text-muted-foreground font-mono mt-0.5">Configure platform parameters, users, tokens, and security</p>
   221→        </div>
   222→        <Badge variant="outline" className="border-primary/30 text-primary font-mono text-xs">CONFIG</Badge>
   223→      </motion.div>
   224→
   225→      <Tabs defaultValue="system" className="space-y-4">
   226→        <TabsList className="bg-muted/50 border border-border/50">
   227→          <TabsTrigger value="system" className="gap-1.5 text-xs"><Settings className="h-3.5 w-3.5" />System</TabsTrigger>
   228→          <TabsTrigger value="users" className="gap-1.5 text-xs"><Users className="h-3.5 w-3.5" />Users</TabsTrigger>
   229→          <TabsTrigger value="tokens" className="gap-1.5 text-xs"><KeyRound className="h-3.5 w-3.5" />API Tokens</TabsTrigger>
   230→          <TabsTrigger value="security" className="gap-1.5 text-xs"><ShieldCheck className="h-3.5 w-3.5" />Security</TabsTrigger>
   231→        </TabsList>
   232→
   233→        {/* System Settings Tab */}
   234→        <TabsContent value="system">
   235→          {settingsLoading ? (
   236→            <div className="space-y-2"><Skeleton className="h-10 w-full" /><Skeleton className="h-10 w-full" /><Skeleton className="h-10 w-full" /></div>
   237→          ) : (
   238→            <Card className="border-border/50 bg-card/80 backdrop-blur-sm">
   239→              <CardContent className="p-0">
   240→                <ScrollArea className="max-h-[500px]">
   241→                  <Table>
   242→                    <TableHeader>
   243→                      <TableRow className="border-border/50 hover:bg-transparent">
   244→                        <TableHead className="text-xs font-mono">Key</TableHead>
   245→                        <TableHead className="text-xs font-mono">Value</TableHead>
   246→                        <TableHead className="text-xs">Category</TableHead>
   247→                        <TableHead className="text-xs text-muted-foreground hidden md:table-cell">Description</TableHead>
   248→                        <TableHead className="w-16"></TableHead>
   249→                      </TableRow>
   250→                    </TableHeader>
   251→                    <TableBody>
   252→                      {Object.entries(groupedSettings).flatMap(([cat, items]) =>
   253→                        items.map((s) => (
   254→                          <TableRow key={s.id} className="border-border/30">
   255→                            <TableCell className="font-mono text-xs text-primary/80 py-2.5">{s.key}</TableCell>
   256→                            <TableCell className="font-mono text-xs py-2.5">
   257→                              {s.isSecret && !revealedKeys.has(s.key)
   258→                                ? <span className="text-muted-foreground">••••••••</span>
   259→                                : <span className="max-w-[200px] truncate inline-block">{s.value}</span>}
   260→                              {s.isSecret && (
   261→                                <button onClick={() => toggleReveal(s.key)} className="ml-2 text-muted-foreground hover:text-foreground">
   262→                                  {revealedKeys.has(s.key) ? <EyeOff className="h-3.5 w-3.5 inline" /> : <Eye className="h-3.5 w-3.5 inline" />}
   263→                                </button>
   264→                              )}
   265→                            </TableCell>
   266→                            <TableCell className="py-2.5">
   267→                              <Badge variant="outline" className={`text-[10px] font-mono ${categoryColors[s.category] || ""}`}>
   268→                                {s.category}
   269→                              </Badge>
   270→                            </TableCell>
   271→                            <TableCell className="text-xs text-muted-foreground hidden md:table-cell py-2.5">{s.description || "—"}</TableCell>
   272→                            <TableCell className="py-2.5">
   273→                              <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => openEdit(s)}>
   274→                                <Pencil className="h-3.5 w-3.5" />
   275→                              </Button>
   276→                            </TableCell>
   277→                          </TableRow>
   278→                        ))
   279→                      )}
   280→                    </TableBody>
   281→                  </Table>
   282→                </ScrollArea>
   283→              </CardContent>
   284→            </Card>
   285→          )}
   286→          <div className="mt-4">
   287→            <p className="text-xs text-muted-foreground mb-2 font-medium">Settings JSON</p>
   288→            <JsonViewer
   289→              data={Object.fromEntries((settings || []).map((s) => [
   290→                s.key,
   291→                { value: s.isSecret && !revealedKeys.has(s.key) ? "••••••••" : s.value, category: s.category, isSecret: s.isSecret, description: s.description },
   292→              ]))}
   293→              maxHeight="max-h-64"
   294→              initialDepth={2}
   295→            />
   296→          </div>
   297→        </TabsContent>
   298→
   299→        {/* Users Tab */}
   300→        <TabsContent value="users">
   301→          <div className="flex justify-end mb-3">
   302→            {isAdmin && <Button size="sm" className="gap-1.5 text-xs" onClick={() => setCreateUserOpen(true)}><Plus className="h-3.5 w-3.5" />Create User</Button>}
   303→          </div>
   304→          <Card className="border-border/50 bg-card/80 backdrop-blur-sm">
   305→            <CardContent className="p-0">
   306→              <ScrollArea className="max-h-[500px]">
   307→                <Table>
   308→                  <TableHeader>
   309→                    <TableRow className="border-border/50 hover:bg-transparent">
   310→                      <TableHead className="text-xs">Email</TableHead>
   311→                      <TableHead className="text-xs">Name</TableHead>
   312→                      <TableHead className="text-xs">Role</TableHead>
   313→                      <TableHead className="text-xs">Active</TableHead>
   314→                      <TableHead className="text-xs font-mono hidden lg:table-cell">Last Login</TableHead>
   315→                      <TableHead className="text-xs font-mono hidden lg:table-cell">Failed</TableHead>
   316→                      <TableHead className="text-xs hidden md:table-cell">Status</TableHead>
   317→                    </TableRow>
   318→                  </TableHeader>
   319→                  <TableBody>
   320→                    {usersLoading ? (
   321→                      Array.from({ length: 3 }).map((_, i) => <TableRow key={i}><TableCell colSpan={7}><Skeleton className="h-8 w-full" /></TableCell></TableRow>)
   322→                    ) : (users || []).map((u) => (
   323→                      <TableRow key={u.id} className="border-border/30">
   324→                        <TableCell className="font-mono text-xs py-2.5">{u.email}</TableCell>
   325→                        <TableCell className="text-xs py-2.5">{u.name || "—"}</TableCell>
   326→                        <TableCell className="py-2.5">
   327→                          {isAdmin ? (
   328→                            <Select value={u.role} onValueChange={(v) => handleRoleChange(u.id, v)}>
   329→                              <SelectTrigger className="h-7 w-28 text-xs"><SelectValue /></SelectTrigger>
   330→                              <SelectContent>
   331→                                <SelectItem value="admin">Admin</SelectItem>
   332→                                <SelectItem value="operator">Operator</SelectItem>
   333→                                <SelectItem value="analyst">Analyst</SelectItem>
   334→                                <SelectItem value="viewer">Viewer</SelectItem>
   335→                              </SelectContent>
   336→                            </Select>
   337→                          ) : (
   338→                            <Badge variant="outline" className={`text-[10px] ${roleColors[u.role] || ""}`}>{u.role}</Badge>
   339→                          )}
   340→                        </TableCell>
   341→                        <TableCell className="py-2.5">
   342→                          <Switch checked={u.isActive} onCheckedChange={() => setDeactivateTarget(u)} disabled={!isAdmin} />
   343→                        </TableCell>
   344→                        <TableCell className="font-mono text-[11px] text-muted-foreground hidden lg:table-cell py-2.5">{fmtDate(u.lastLoginAt)}</TableCell>
   345→                        <TableCell className="font-mono text-[11px] hidden lg:table-cell py-2.5">
   346→                          <span className={u.failedAttempts > 3 ? "text-red-400" : "text-muted-foreground"}>{u.failedAttempts}</span>
   347→                        </TableCell>
   348→                        <TableCell className="hidden md:table-cell py-2.5">
   349→                          {u.lockedUntil && new Date(u.lockedUntil) > new Date() ? (
   350→                            <Badge variant="outline" className="text-[10px] bg-red-500/10 text-red-400 border-red-500/20">
   351→                              <Lock className="h-3 w-3 mr-1" />Locked
   352→                            </Badge>
   353→                          ) : (
   354→                            <Badge variant="outline" className="text-[10px] bg-emerald-500/10 text-emerald-400 border-emerald-500/20">OK</Badge>
   355→                          )}
   356→                        </TableCell>
   357→                      </TableRow>
   358→                    ))}
   359→                  </TableBody>
   360→                </Table>
   361→              </ScrollArea>
   362→            </CardContent>
   363→          </Card>
   364→        </TabsContent>
   365→
   366→        {/* API Tokens Tab */}
   367→        <TabsContent value="tokens">
   368→          <div className="flex justify-end mb-3">
   369→            <Button size="sm" className="gap-1.5 text-xs" onClick={() => { setCreateTokenOpen(true); setCreatedToken(null); }}>
   370→              <Plus className="h-3.5 w-3.5" />Create Token
   371→            </Button>
   372→          </div>
   373→          <Card className="border-border/50 bg-card/80 backdrop-blur-sm">
   374→            <CardContent className="p-0">
   375→              <ScrollArea className="max-h-[500px]">
   376→                <Table>
   377→                  <TableHeader>
   378→                    <TableRow className="border-border/50 hover:bg-transparent">
   379→                      <TableHead className="text-xs">Name</TableHead>
   380→                      <TableHead className="text-xs hidden md:table-cell">Purpose</TableHead>
   381→                      <TableHead className="text-xs font-mono">Created</TableHead>
   382→                      <TableHead className="text-xs font-mono hidden lg:table-cell">Expires</TableHead>
   383→                      <TableHead className="text-xs font-mono hidden lg:table-cell">Last Used</TableHead>
   384→                      <TableHead className="text-xs">Status</TableHead>
   385→                      <TableHead className="w-16"></TableHead>
   386→                    </TableRow>
   387→                  </TableHeader>
   388→                  <TableBody>
   389→                    {tokensLoading ? (
   390→                      Array.from({ length: 3 }).map((_, i) => <TableRow key={i}><TableCell colSpan={7}><Skeleton className="h-8 w-full" /></TableCell></TableRow>)
   391→                    ) : (tokens || []).map((t) => (
   392→                      <TableRow key={t.id} className="border-border/30">
   393→                        <TableCell className="text-xs font-medium py-2.5">{t.name}</TableCell>
   394→                        <TableCell className="text-xs text-muted-foreground hidden md:table-cell py-2.5">{t.purpose || "—"}</TableCell>
   395→                        <TableCell className="font-mono text-[11px] text-muted-foreground py-2.5">{fmtDate(t.createdAt)}</TableCell>
   396→                        <TableCell className="font-mono text-[11px] text-muted-foreground hidden lg:table-cell py-2.5">{fmtDate(t.expiresAt)}</TableCell>
   397→                        <TableCell className="font-mono text-[11px] text-muted-foreground hidden lg:table-cell py-2.5">{fmtDate(t.lastUsedAt)}</TableCell>
   398→                        <TableCell className="py-2.5">
   399→                          {t.isRevoked ? (
   400→                            <Badge variant="outline" className="text-[10px] bg-red-500/10 text-red-400 border-red-500/20">Revoked</Badge>
   401→                          ) : (
   402→                            <Badge variant="outline" className="text-[10px] bg-emerald-500/10 text-emerald-400 border-emerald-500/20">Active</Badge>
   403→                          )}
   404→                        </TableCell>
   405→                        <TableCell className="py-2.5">
   406→                          {!t.isRevoked && (
   407→                            <Button variant="ghost" size="icon" className="h-7 w-7 text-red-400 hover:text-red-300 hover:bg-red-500/10" onClick={() => handleRevokeToken(t.id)}>
   408→                              <Trash2 className="h-3.5 w-3.5" />
   409→                            </Button>
   410→                          )}
   411→                        </TableCell>
   412→                      </TableRow>
   413→                    ))}
   414→                  </TableBody>
   415→                </Table>
   416→              </ScrollArea>
   417→            </CardContent>
   418→          </Card>
   419→        </TabsContent>
   420→
   421→        {/* API Tokens Tab */}
   422→        <TabsContent value="tokens">
   423→          <div className="flex justify-end mb-3">
   424→            <Button size="sm" className="gap-1.5 text-xs" onClick={() => { setCreateTokenOpen(true); setCreatedToken(null); }}>
   425→              <Plus className="h-3.5 w-3.5" />Create Token
   426→            </Button>
   427→          </div>
   428→          <Card className="border-border/50 bg-card/80 backdrop-blur-sm">
   429→            <CardContent className="p-0">
   430→              <ScrollArea className="max-h-[500px]">
   431→                <Table>
   432→                  <TableHeader>
